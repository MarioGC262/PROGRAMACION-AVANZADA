3. División de los Datos (80% Entrenamiento, 20% Prueba)

set.seed(123) # Para reproducibilidad
n <- nrow(datos)
indices_entrenamiento <- sample(1:n, size = round(0.8 * n))

train_data <- datos[indices_entrenamiento, ]
test_data <- datos[-indices_entrenamiento, ]

cat("Tamaño del set de entrenamiento:", nrow(train_data), "\n")

5. Predicción y Matriz de Confusión

predicciones <- predict(modelo_nb, test_data)

# Matriz de Confusión
matriz_confusion <- table(Prediccion = predicciones, Real = test_data$Rendimiento)
print(matriz_confusion)
